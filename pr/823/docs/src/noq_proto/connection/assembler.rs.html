7/// Helper to assemble unordered stream frames into an ordered stream
8#[derive(Debug, Default)]
9pub(super) struct Assembler {
10    state: State,
11    /// Buffered chunks, in `Buffer::cmp` order
12    data: VecDeque<Buffer>,
13    /// Total number of buffered bytes, including duplicates in ordered mode.
14    buffered: usize,
15    /// Estimated number of allocated bytes, will never be less than `buffered`.
16    allocated: usize,
17    /// Number of bytes read by the application. When only ordered reads have been used, this is
18    /// the length of the contiguous prefix of the stream which has been consumed by the
19    /// application, aka the stream offset.
20    bytes_read: u64,
21    end: u64,
22}
23
24impl Assembler {
25    pub(super) fn new() -> Self {
26        Self::default()
27    }
28
29    /// Reset to the initial state
30    pub(super) fn reinit(&mut self) {
31        let old_data = mem::take(&mut self.data);
32        *self = Self::default();
33        self.data = old_data;
34        self.data.clear();
35    }
36
37    pub(super) fn is_ordered(&self) -> bool {
38        self.state.is_ordered()
39    }
40
41    pub(super) fn ensure_ordering(&mut self, ordered: bool) -> Result<(), IllegalOrderedRead> {
42        if ordered && !self.state.is_ordered() {
43            return Err(IllegalOrderedRead);
44        } else if !ordered && self.state.is_ordered() {
45            // Enter unordered mode
46            if !self.data.is_empty() {
47                // Get rid of possible duplicates
48                self.defragment();
49            }
50            let mut recvd = ArrayRangeSet::new();
51            recvd.insert(0..self.bytes_read);
52            for chunk in &self.data {
53                recvd.insert(chunk.offset..chunk.offset + chunk.bytes.len() as u64);
54            }
55            self.state = State::Unordered { recvd };
56        }
57        Ok(())
58    }
59
60    /// Get the the next chunk
61    pub(super) fn read(&mut self, max_length: usize, ordered: bool) -> Option<Chunk> {
62        loop {
63            let chunk = self.data.front_mut()?;
64
65            if ordered {
66                if chunk.offset > self.bytes_read {
67                    // Next chunk is after current read index
68                    return None;
69                } else if (chunk.offset + chunk.bytes.len() as u64) <= self.bytes_read {
70                    // Next chunk is useless as the read index is beyond its end
71                    self.buffered -= chunk.bytes.len();
72                    self.allocated -= chunk.allocation_size;
73                    self.data.pop_front();
74                    continue;
75                }
76
77                // Determine `start` and `len` of the slice of useful data in chunk
78                let start = (self.bytes_read - chunk.offset) as usize;
79                // Advancing the offset can push the front past data[1]; both exits below fix that
80                if start > 0 {
81                    chunk.bytes.advance(start);
82                    chunk.offset += start as u64;
83                    self.buffered -= start;
84                }
85            }
86
87            if max_length < chunk.bytes.len() {
88                self.bytes_read += max_length as u64;
89                let offset = chunk.offset;
90                chunk.offset += max_length as u64;
91                self.buffered -= max_length;
92                let bytes = chunk.bytes.split_to(max_length);
93                self.restore_front_order();
94                return Some(Chunk::new(offset, bytes));
95            }
96
97            self.bytes_read += chunk.bytes.len() as u64;
98            self.buffered -= chunk.bytes.len();
99            self.allocated -= chunk.allocation_size;
100            let offset = chunk.offset;
101            let bytes = mem::take(&mut chunk.bytes);
102            self.data.pop_front();
103            return Some(Chunk::new(offset, bytes));
104        }
105    }
106
107    /// Restore ordering after `read` advanced the front's offset
108    fn restore_front_order(&mut self) {
109        if self.data.len() < 2 || self.data[0] <= self.data[1] {
110            return;
111        }
112        let chunk = self.data.pop_front().unwrap();
113        let idx = self.data.partition_point(|other| *other <= chunk);
114        self.data.insert(idx, chunk);
115    }

117    /// Copy fragmented chunk data to new chunks backed by a single buffer
118    ///
119    /// This makes sure we're not unnecessarily holding on to many larger allocations.
120    /// We merge contiguous chunks in the process of doing so.
121    fn defragment(&mut self) {
122        let mut buffers = mem::take(&mut self.data);
123        self.buffered = 0;
124        let mut fragmented_buffered = 0;
125        let mut offset = self.bytes_read;
126        for chunk in &mut buffers {
127            chunk.try_mark_defragment(offset);
128            let size = chunk.bytes.len();
129            offset = chunk.offset + size as u64;
130            self.buffered += size;
131            if !chunk.defragmented {
132                fragmented_buffered += size;
133            }
134        }
135        self.allocated = self.buffered;
136        self.data.reserve(buffers.len());
137        let mut buffer = BytesMut::with_capacity(fragmented_buffered);
138        let mut offset = self.bytes_read;
139        for chunk in buffers {
140            if chunk.defragmented {
141                // bytes might be empty after try_mark_defragment
142                if chunk.bytes.is_empty() {
143                    continue;
144                }
145                // The accumulated chunk starts before this one, so it goes first
146                if !buffer.is_empty() {
147                    self.data
148                        .push_back(Buffer::new_defragmented(offset, buffer.split().freeze()));
149                }
150                offset = chunk.offset + chunk.bytes.len() as u64;
151                self.data.push_back(chunk);
152                continue;
153            }
154            // Overlap is resolved by try_mark_defragment
155            if chunk.offset != offset + (buffer.len() as u64) {
156                if !buffer.is_empty() {
157                    self.data
158                        .push_back(Buffer::new_defragmented(offset, buffer.split().freeze()));
159                }
160                offset = chunk.offset;
161            }
162            buffer.extend_from_slice(&chunk.bytes);
163        }
164        if !buffer.is_empty() {
165            self.data
166                .push_back(Buffer::new_defragmented(offset, buffer.split().freeze()));
167        }
168    }

170    // Note: If a packet contains many frames from the same stream, the estimated over-allocation
171    // will be much higher because we are counting the same allocation multiple times.
172    pub(super) fn insert(
173        &mut self,
174        mut offset: u64,
175        mut bytes: Bytes,
176        allocation_size: usize,
177    ) -> Result<(), TooManyChunks> {
178        debug_assert!(
179            bytes.len() <= allocation_size,
180            "allocation_size less than bytes.len(): {:?} < {:?}",
181            allocation_size,
182            bytes.len()
183        );
184        self.end = self.end.max(offset + bytes.len() as u64);
185        if let State::Unordered { ref mut recvd } = self.state {
186            // Discard duplicate data
187            let range = offset..offset + bytes.len() as u64;
188            for duplicate in recvd.iter_range(range.clone()) {
189                if duplicate.start > offset {
190                    let buffer = Buffer::new(
191                        offset,
192                        bytes.split_to((duplicate.start - offset) as usize),
193                        allocation_size,
194                    );
195                    self.buffered += buffer.bytes.len();
196                    self.allocated += buffer.allocation_size;
197                    insert_ordered(&mut self.data, buffer);
198                    offset = duplicate.start;
199                }
200                bytes.advance((duplicate.end - offset) as usize);
201                offset = duplicate.end;
202            }
203            recvd.insert(range);
204        } else if offset < self.bytes_read {
205            if (offset + bytes.len() as u64) <= self.bytes_read {
206                return Ok(());
207            } else {
208                let diff = self.bytes_read - offset;
209                offset += diff;
210                bytes.advance(diff as usize);
211            }
212        }
213
214        // No early return when empty: the dedup loop above may already have pushed chunks.
215        if !bytes.is_empty() {
216            let buffer = Buffer::new(offset, bytes, allocation_size);
217            self.buffered += buffer.bytes.len();
218            self.allocated += buffer.allocation_size;
219            insert_ordered(&mut self.data, buffer);
220        }
221        // `self.buffered` also counts duplicate bytes, therefore we use
222        // `self.end - self.bytes_read` as an upper bound of buffered unique
223        // bytes. This will cause a defragmentation if the amount of duplicate
224        // bytes exceedes a proportion of the receive window size.
225        let buffered = self.buffered.min((self.end - self.bytes_read) as usize);
226        let over_allocation = self.allocated - buffered;
227        // Rationale: on the one hand, we want to defragment rarely, ideally never
228        // in non-pathological scenarios. However, a pathological or malicious
229        // peer could send us one-byte frames, and since we use reference-counted
230        // buffers in order to prevent copying, this could result in keeping a lot
231        // of memory allocated. This limits over-allocation in proportion to the
232        // buffered data. The constants are chosen somewhat arbitrarily and try to
233        // balance between defragmentation overhead and over-allocation.
234        let threshold = 32768.max(buffered * 3 / 2);
235        // Small gapped frames hold over-allocation below the threshold, so bound the count too.
236        if over_allocation > threshold || self.data.len() > COMPACT_THRESHOLD {
237            self.defragment();
238            // ngtcp2 uses a threshold of 4000 -- try to be a little more conservative?
239            if self.data.len() > MAX_CHUNKS {
240                return Err(TooManyChunks);
241            }
242        }
243
244        Ok(())
245    }

248    pub(super) fn bytes_read(&self) -> u64 {
249        self.bytes_read
250    }

252    /// Discard all buffered data
253    pub(super) fn clear(&mut self) {
254        self.data.clear();

256        self.allocated = 0;
257    }

275#[derive(Debug, Eq)]
276struct Buffer {
277    offset: u64,
278    bytes: Bytes,
279    /// Size of the allocation behind `bytes`, if `defragmented == false`.
280    /// Otherwise this will be set to `bytes.len()` by `try_mark_defragment`.
281    /// Will never be less than `bytes.len()`.
282    allocation_size: usize,
283    defragmented: bool,
284}
285
286impl Buffer {
287    /// Constructs a new fragmented Buffer
288    fn new(offset: u64, bytes: Bytes, allocation_size: usize) -> Self {
289        Self {
290            offset,
291            bytes,
292            allocation_size,
293            defragmented: false,
294        }
295    }
296
297    /// Constructs a new defragmented Buffer
298    fn new_defragmented(offset: u64, bytes: Bytes) -> Self {
299        let allocation_size = bytes.len();
300        Self {
301            offset,
302            bytes,
303            allocation_size,
304            defragmented: true,
305        }
306    }
307
308    /// Discards data before `offset` and flags `self` as defragmented if it has good utilization
309    fn try_mark_defragment(&mut self, offset: u64) {
310        let duplicate = offset.saturating_sub(self.offset) as usize;
311        self.offset = self.offset.max(offset);
312        if duplicate >= self.bytes.len() {
313            // All bytes are duplicate
314            self.bytes = Bytes::new();
315            self.defragmented = true;
316            self.allocation_size = 0;
317            return;
318        }
319        self.bytes.advance(duplicate);
320        // Make sure that fragmented buffers with high utilization become defragmented and
321        // defragmented buffers remain defragmented
322        self.defragmented = self.defragmented || self.bytes.len() * 6 / 5 >= self.allocation_size;
323        if self.defragmented {
324            // Make sure that defragmented buffers do not contribute to over-allocation
325            self.allocation_size = self.bytes.len();
326        }
327    }
328}

330impl Ord for Buffer {
331    // Ascending offset, longer chunks first at equal offsets
332    fn cmp(&self, other: &Self) -> Ordering {
333        self.offset
334            .cmp(&other.offset)
335            .then(other.bytes.len().cmp(&self.bytes.len()))
336    }
337}
338
339impl PartialOrd for Buffer {
340    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
341        Some(self.cmp(other))
342    }
343}
344
345impl PartialEq for Buffer {
346    fn eq(&self, other: &Self) -> bool {
347        (self.offset, self.bytes.len()) == (other.offset, other.bytes.len())
348    }
349}

376/// O(1) for a frame in order or below everything buffered, otherwise an insert that
377/// shifts up to half of `data` (see `COMPACT_THRESHOLD`)
378fn insert_ordered(data: &mut VecDeque<Buffer>, buffer: Buffer) {
379    if data.back().is_none_or(|back| *back <= buffer) {
380        data.push_back(buffer);
381        return;
382    }
383    // Order among equal chunks is unspecified, as it was under the heap, so a duplicate
384    // of the front takes this path rather than an insert
385    if data.front().is_some_and(|front| buffer <= *front) {
386        data.push_front(buffer);
387        return;
388    }
389    let idx = data.partition_point(|chunk| *chunk <= buffer);
390    data.insert(idx, buffer);
391}

393/// Bound on the number of distinct spans kept for a stream
394///
395/// Independent of how much memory those spans over-allocate. A frame is rejected only
396/// if compaction cannot get the count back down to this.
397const MAX_CHUNKS: usize = 1024;
398
399/// Chunk count past which `insert` compacts before deciding whether to reject
400///
401/// Above `MAX_CHUNKS` so a flood of mergeable frames cannot force a defragmentation
402/// per frame.
403const COMPACT_THRESHOLD: usize = 2 * MAX_CHUNKS;

834    /// Fragments arriving retransmitted and shuffled must still read back as the exact stream
835    fn random_stream(ordered: bool) {
836        const LEN: usize = 2048;
837        let mut rng = new_rng();
838        let stream: Vec<u8> = (0..LEN).map(|_| rng.random()).collect();
839        for _ in 0..64 {
840            let mut frags = Vec::new();
841            let mut at = 0;
842            while at < LEN {
843                let size = rng.random_range(1..=32).min(LEN - at);
844                frags.push((at as u64, size));
845                at += size;
846            }
847            // Retransmits are refragmented from scratch, so they overlap the original frames at
848            // arbitrary boundaries rather than repeating them
849            for _ in 0..frags.len() / 4 {
850                let start = rng.random_range(0..LEN);
851                let end = (start + rng.random_range(1..=64)).min(LEN);
852                let mut at = start;
853                while at < end {
854                    let size = rng.random_range(1..=32).min(end - at);
855                    frags.push((at as u64, size));
856                    at += size;
857                }
858            }
859            frags.shuffle(&mut rng);
860
861            let mut x = Assembler::new();
862            if !ordered {
863                x.ensure_ordering(false).unwrap();
864            }
865            let mut delivered = vec![false; LEN];
866            for (offset, size) in frags {
867                let bytes =
868                    Bytes::copy_from_slice(&stream[offset as usize..offset as usize + size]);
869                x.insert(offset, bytes, size).unwrap();
870                assert_sorted(&x);
871                if rng.random_ratio(1, 3) {
872                    let max = rng.random_range(1..=64);
873                    drain(&mut x, max, ordered, &stream, &mut delivered);
874                }
875            }
876            drain(&mut x, usize::MAX, ordered, &stream, &mut delivered);
877            assert!(
878                delivered.iter().all(|d| *d),
879                "stream did not fully assemble"
880            );
881        }
882    }

884    /// Read `x` dry, checking chunks against `stream` and that no byte arrives twice
885    #[track_caller]
886    fn drain(x: &mut Assembler, max: usize, ordered: bool, stream: &[u8], delivered: &mut [bool]) {
887        while let Some(chunk) = x.read(max, ordered) {
888            let start = chunk.offset as usize;
889            let end = start + chunk.bytes.len();
890            assert_eq!(&chunk.bytes[..], &stream[start..end]);
891            for d in &mut delivered[start..end] {
892                assert!(!*d, "offset {start} delivered twice");
893                *d = true;
894            }
895            assert_sorted(x);
896        }
897    }

941    use proptest::prelude::*;

947    const MAX_OFFSET: u64 = 512;
948    const MAX_LEN: usize = 64;
949
950    #[derive(Debug, Clone, Arbitrary)]
951    enum Op {
952        #[weight(10)]
953        Insert {
954            #[strategy(0..MAX_OFFSET)]
955            offset: u64,
956            #[strategy(1..MAX_LEN)]
957            len: usize,
958        },
959        #[weight(10)]
960        Read {
961            #[strategy(1..MAX_LEN)]
962            max_len: usize,
963        },
964        #[weight(1)]
965        EnsureOrdering { ordered: bool },
966        #[weight(1)]
967        Defragment,
968    }
969
970    /// Tracks the state of the assembler for verification
971    struct RefState {
972        received: Vec<bool>,
973        returned: Vec<bool>,
974        ordered: bool,
975    }
976
977    fn set_range(bits: &mut [bool], start: u64, len: usize) {
978        for i in start..(start + len as u64).min(bits.len() as u64) {
979            bits[i as usize] = true;
980        }
981    }
982
983    impl RefState {
984        fn new() -> Self {
985            Self {
986                received: vec![false; MAX_OFFSET as usize],
987                returned: vec![false; MAX_OFFSET as usize],
988                ordered: true,
989            }
990        }
991
992        fn insert(&mut self, offset: u64, len: usize) {
993            set_range(&mut self.received, offset, len);
994        }
995
996        fn ensure_ordering(&mut self, ordered: bool) -> bool {
997            if ordered && !self.ordered {
998                return false;
999            }
1000            self.ordered = ordered;
1001            true
1002        }
1003
1004        fn bytes_read(&self) -> u64 {
1005            self.returned.iter().filter(|&&x| x).count() as u64
1006        }
1007    }
1008
1009    fn make_data() -> Vec<u8> {
1010        use rand::SeedableRng;
1011        let mut rng = rand::rngs::StdRng::seed_from_u64(0xDEADBEEF);
1012        let mut data = vec![0u8; MAX_OFFSET as usize];
1013        rng.fill(data.as_mut_slice());
1014        data
1015    }
1016
1017    fn get_slice(data: &[u8], offset: u64, len: usize) -> Bytes {
1018        let start = offset as usize;
1019        let end = (start + len).min(data.len());
1020        Bytes::copy_from_slice(&data[start..end])
1021    }
1022
1023    fn verify_chunk(data: &[u8], chunk: &Chunk) -> bool {
1024        let start = chunk.offset as usize;
1025        chunk.bytes[..] == data[start..start + chunk.bytes.len()]
1026    }
1027
1028    #[proptest]
1029    fn assembler_matches_reference(
1030        #[strategy(proptest::collection::vec(any::<Op>(), 1..100))] ops: Vec<Op>,
1031    ) {
1032        let data = make_data();
1033        let mut asm = Assembler::new();
1034        let mut reference = RefState::new();
1035
1036        for op in ops {
1037            match op {
1038                Op::Insert { offset, len } => {
1039                    let bytes = get_slice(&data, offset, len);
1040                    asm.insert(offset, bytes, len).unwrap();
1041                    reference.insert(offset, len);
1042                }
1043                Op::Read { max_len } => {
1044                    let ordered = reference.ordered;
1045                    let actual = asm.read(max_len, ordered);
1046
1047                    match actual {
1048                        None => {
1049                            // Should only be None if no unreturned received bytes available
1050                            let has_available = if ordered {
1051                                // In ordered mode, check if the first unreturned byte is received
1052                                reference
1053                                    .returned
1054                                    .iter()
1055                                    .position(|&x| !x)
1056                                    .is_some_and(|pos| reference.received[pos])
1057                            } else {
1058                                // In unordered mode, check if any unreturned received byte exists
1059                                reference
1060                                    .received
1061                                    .iter()
1062                                    .zip(&reference.returned)
1063                                    .any(|(&r, &ret)| r && !ret)
1064                            };
1065                            prop_assert!(
1066                                !has_available,
1067                                "read returned None but data was available"
1068                            );
1069                        }
1070                        Some(chunk) => {
1071                            prop_assert!(chunk.bytes.len() <= max_len, "chunk exceeds max_len");
1072                            prop_assert!(verify_chunk(&data, &chunk), "data corruption");
1073                            // Mark as returned, check for duplicates
1074                            for i in 0..chunk.bytes.len() {
1075                                let offset = chunk.offset as usize + i;
1076                                prop_assert!(
1077                                    reference.received[offset],
1078                                    "returned unreceived byte at {offset}"
1079                                );
1080                                prop_assert!(
1081                                    !reference.returned[offset],
1082                                    "duplicate byte at {offset}"
1083                                );
1084                                reference.returned[offset] = true;
1085                            }
1086                        }
1087                    }
1088                }
1089                Op::EnsureOrdering { ordered } => {
1090                    let actual = asm.ensure_ordering(ordered).is_ok();
1091                    let expected = reference.ensure_ordering(ordered);
1092                    prop_assert_eq!(actual, expected, "ensure_ordering result mismatch");
1093                }
1094                Op::Defragment => {
1095                    if asm.state.is_ordered() {
1096                        asm.defragment();
1097                    }
1098                }
1099            }
1100        }
1101
1102        // Invariant: bytes_read matches
1103        prop_assert_eq!(
1104            asm.bytes_read(),
1105            reference.bytes_read(),
1106            "bytes_read mismatch"
1107        );
1108    }
