18/// Abstracts I/O and timer operations for runtime independence
19pub trait Runtime: Send + Sync + Debug + 'static {
20    /// Construct a timer that will expire at `i`
21    fn new_timer(&self, i: Instant) -> Pin<Box<dyn AsyncTimer>>;
22    /// Drive `future` to completion in the background
23    #[track_caller]
24    fn spawn(&self, future: Pin<Box<dyn Future<Output = ()> + Send>>);
25    /// Convert `t` into the socket type used by this runtime
26    #[cfg(not(wasm_browser))]
27    fn wrap_udp_socket(&self, t: std::net::UdpSocket) -> io::Result<Box<dyn AsyncUdpSocket>>;
28    /// Look up the current time
29    ///
30    /// Allows simulating the flow of time for testing.
31    fn now(&self) -> Instant {
32        Instant::now()
33    }
34}

36/// Abstract implementation of an async timer for runtime independence
37pub trait AsyncTimer: Send + Debug + 'static {
38    /// Update the timer to expire at `i`
39    fn reset(self: Pin<&mut Self>, i: Instant);
40    /// Check whether the timer has expired, and register to be woken if not
41    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<()>;
42}

44/// Abstract implementation of a UDP socket for runtime independence
45pub trait AsyncUdpSocket: Send + Sync + Debug + 'static {
46    /// Create a [`UdpSender`] that can register a single task for write-readiness notifications
47    /// and send a transmit, if ready.
48    ///
49    /// A `poll_send` method on a single object can usually store only one [`Waker`] at a time,
50    /// i.e. allow at most one caller to wait for an event. This method allows any number of
51    /// interested tasks to construct their own [`UdpSender`] object. They can all then wait for the
52    /// same event and be notified concurrently, because each [`UdpSender`] can store a separate
53    /// [`Waker`].
54    ///
55    /// [`Waker`]: std::task::Waker
56    fn create_sender(&self) -> Pin<Box<dyn UdpSender>>;
57
58    /// Receive UDP datagrams, or register to be woken if receiving may succeed in the future
59    fn poll_recv(
60        &mut self,
61        cx: &mut Context<'_>,
62        bufs: &mut [IoSliceMut<'_>],
63        meta: &mut [RecvMeta],
64    ) -> Poll<io::Result<usize>>;
65
66    /// Look up the local IP address and port used by this socket
67    fn local_addr(&self) -> io::Result<SocketAddr>;
68
69    /// Maximum number of datagrams that might be described by a single [`RecvMeta`]
70    fn max_receive_segments(&self) -> NonZeroUsize {
71        NonZeroUsize::MIN
72    }
73
74    /// Whether datagrams might get fragmented into multiple parts
75    ///
76    /// Sockets should prevent this for best performance. See e.g. the `IPV6_DONTFRAG` socket
77    /// option.
78    fn may_fragment(&self) -> bool {
79        true
80    }
81}

83/// An object for asynchronously writing to an associated [`AsyncUdpSocket`].
84///
85/// Any number of [`UdpSender`]s may exist for a single [`AsyncUdpSocket`]. Each [`UdpSender`] is
86/// responsible for notifying at most one task for send readiness.
87pub trait UdpSender: Send + Sync + Debug + 'static {
88    /// Send a UDP datagram, or register to be woken if sending may succeed in the future.
89    ///
90    /// Usually implementations of this will poll the socket for writability before trying to
91    /// write to them, and retry both if writing fails.
92    ///
93    /// noq will create multiple [`UdpSender`]s, one for each task it's using it from. Thus it's
94    /// important to poll the underlying socket in a way that doesn't overwrite wakers.
95    ///
96    /// A single [`UdpSender`] will be reused, even if `poll_send` returns `Poll::Ready` once,
97    /// unlike [`Future::poll`], so calling it again after readiness should not panic.
98    fn poll_send(
99        self: Pin<&mut Self>,
100        transmit: &Transmit<'_>,
101        cx: &mut Context<'_>,
102    ) -> Poll<io::Result<()>>;
103
104    /// Maximum number of datagrams that a [`Transmit`] may encode.
105    fn max_transmit_segments(&self) -> NonZeroUsize {
106        NonZeroUsize::MIN
107    }
108}

110pin_project_lite::pin_project! {
111    /// A helper for constructing [`UdpSender`]s from an underlying `Socket` type.
112    ///
113    /// This struct implements [`UdpSender`] if `MakeWritableFn` produces a `WritableFut`.
114    ///
115    /// Also serves as a trick, since `WritableFut` doesn't need to be a named future,
116    /// it can be an anonymous async block, as long as `MakeWritableFn` produces that
117    /// anonymous async block type.
118    ///
119    /// The `UdpSenderHelper` generic type parameters don't need to named, as it will be
120    /// used in its dyn-compatible form as a `Pin<Box<dyn UdpSender>>`.
121    struct UdpSenderHelper<Socket, MakeWritableFutFn, WritableFut> {
122        socket: Socket,
123        last_send_error: Option<Instant>,
124        make_writable_fut_fn: MakeWritableFutFn,
125        #[pin]
126        writable_fut: Option<WritableFut>,
127    }
128}

131    for UdpSenderHelper<Socket, MakeWritableFutFn, WritableFut>
132{
133    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
134        f.write_str("UdpSender")
135    }

138impl<Socket, MakeWritableFutFn, WriteableFut>
139    UdpSenderHelper<Socket, MakeWritableFutFn, WriteableFut>
140{
141    /// Create helper that implements [`UdpSender`] from a socket.
142    ///
143    /// Additionally you need to provide what is essentially an async function
144    /// that resolves once the socket is write-ready.
145    ///
146    /// See also the bounds on this struct's [`UdpSender`] implementation.
147    #[cfg(any(feature = "runtime-smol", feature = "runtime-tokio",))]
148    fn new(inner: Socket, make_fut: MakeWritableFutFn) -> Self {
149        Self {
150            socket: inner,
151            last_send_error: None,
152            make_writable_fut_fn: make_fut,
153            writable_fut: None,
154        }
155    }
156}
157
158impl<Socket, MakeWritableFutFn, WritableFut> UdpSender
159    for UdpSenderHelper<Socket, MakeWritableFutFn, WritableFut>
160where
161    Socket: UdpSenderHelperSocket,
162    MakeWritableFutFn: Fn(&Socket) -> WritableFut + Send + Sync + 'static,
163    WritableFut: Future<Output = io::Result<()>> + Send + Sync + 'static,
164{
165    fn poll_send(
166        self: Pin<&mut Self>,
167        transmit: &Transmit<'_>,
168        cx: &mut Context<'_>,
169    ) -> Poll<io::Result<()>> {
170        let mut this = self.project();
171        loop {
172            if this.writable_fut.is_none() {
173                this.writable_fut
174                    .set(Some((this.make_writable_fut_fn)(this.socket)));
175            }
176            // We're forced to `unwrap` here because `Fut` may be `!Unpin`, which means we can't
177            // safely obtain an `&mut WritableFut` after storing it in
178            // `self.writable_fut` when `self` is already behind `Pin`, and if we didn't
179            // store it then we wouldn't be able to keep it alive between `poll_send`
180            // calls.
181            let result =
182                std::task::ready!(this.writable_fut.as_mut().as_pin_mut().unwrap().poll(cx));
183
184            // Polling an arbitrary `Future` after it becomes ready is a logic error, so arrange for
185            // a new `Future` to be created on the next call.
186            this.writable_fut.set(None);
187
188            // If .writable() fails, propagate the error
189            result?;
190
191            match this.socket.try_send(transmit) {
192                // We thought the socket was writable, but it wasn't, then retry so that either
193                // another `writable().await` call determines that the socket is
194                // indeed not writable and registers us for a wakeup, or the send
195                // succeeds if this really was just a transient failure.
196                Err(e) if e.kind() == io::ErrorKind::WouldBlock => continue,
197                Err(e) => {
198                    log_sendmsg_error(this.last_send_error, &e, transmit);
199                    return Poll::Ready(Ok(()));
200                }
201                Ok(()) => return Poll::Ready(Ok(())),
202            }
203        }
204    }
205
206    fn max_transmit_segments(&self) -> NonZeroUsize {
207        self.socket.max_transmit_segments()
208    }
209}

211/// Limits I/O error logging to one message per minute.
212const IO_ERROR_LOG_INTERVAL: Duration = Duration::from_secs(60);
213
214fn log_sendmsg_error(
215    last_send_error: &mut Option<Instant>,
216    error: &io::Error,
217    transmit: &Transmit<'_>,
218) {
219    #[cfg(unix)]
220    // Unix `EMSGSIZE` is expected for MTU probes.
221    if udp::is_msg_size_err(error) {
222        return;
223    }
224
225    let now = Instant::now();
226    if last_send_error
227        .is_some_and(|last| now.saturating_duration_since(last) <= IO_ERROR_LOG_INTERVAL)
228    {
229        return;
230    }
231    *last_send_error = Some(now);
232
233    tracing::warn!(
234        "sendmsg error: {:?}, Transmit: {{ destination: {:?}, src_ip: {:?}, ecn: {:?}, len: {:?}, segment_size: {:?} }}",
235        error,
236        transmit.destination,
237        transmit.src_ip,
238        transmit.ecn,
239        transmit.contents.len(),
240        transmit.segment_size
241    );
242}

244/// Parts of the [`UdpSender`] trait that aren't asynchronous or require storing wakers.
245///
246/// This trait is used by [`UdpSenderHelper`] to help construct [`UdpSender`]s.
247trait UdpSenderHelperSocket: Send + Sync + 'static {
248    /// Try to send a transmit, if the socket happens to be write-ready.
249    ///
250    /// If not write-ready, this is allowed to return [`std::io::ErrorKind::WouldBlock`].
251    ///
252    /// The [`UdpSenderHelper`] will use this to implement [`UdpSender::poll_send`].
253    fn try_send(&self, transmit: &Transmit<'_>) -> io::Result<()>;
254
255    /// See [`UdpSender::max_transmit_segments`].
256    fn max_transmit_segments(&self) -> NonZeroUsize;
257}

259/// Automatically select an appropriate runtime from those enabled at compile time
260///
261/// If `runtime-tokio` is enabled and this function is called from within a Tokio runtime context,
262/// then `TokioRuntime` is returned. Otherwise, if `runtime-smol` is enabled, `SmolRuntime` is
263/// returned. Otherwise, `None` is returned.
264#[cfg(any(feature = "runtime-tokio", feature = "runtime-smol"))]
265#[allow(clippy::needless_return)] // Be sure we return the right thing
266pub fn default_runtime() -> Option<Arc<dyn Runtime>> {
267    #[cfg(feature = "runtime-tokio")]
268    {
269        if ::tokio::runtime::Handle::try_current().is_ok() {
270            return Some(Arc::new(TokioRuntime));
271        }
272    }
273
274    #[cfg(feature = "runtime-smol")]
275    {
276        return Some(Arc::new(SmolRuntime));
277    }
278
279    #[cfg(not(feature = "runtime-smol"))]
280    None
281}
