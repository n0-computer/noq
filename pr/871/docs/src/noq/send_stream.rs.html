21/// A stream that can only be used to send data
22///
23/// If dropped, streams that haven't been explicitly [`reset()`] will be implicitly [`finish()`]ed,
24/// continuing to (re)transmit previously written data until it has been fully acknowledged or the
25/// connection is closed.
26///
27/// # Cancellation
28///
29/// A `write` method is said to be *cancel-safe* when dropping its future before the future becomes
30/// ready will always result in no data being written to the stream. This is true of methods which
31/// succeed immediately when any progress is made, and is not true of methods which might need to
32/// perform multiple writes internally before succeeding. Each `write` method documents whether it
33/// is cancel-safe.
34///
35/// [`reset()`]: SendStream::reset
36/// [`finish()`]: SendStream::finish
37#[derive(derive_more::Debug)]
38pub struct SendStream {
39    conn: ConnectionRef,
40    stream: StreamId,
41    is_0rtt: bool,
42    #[debug("{}", StreamAffinityCallbackOption::new(affinity_cb))]
43    affinity_cb: Option<Arc<dyn StreamAffinityCallback>>,
44}

56    /// Write a buffer into this stream, returning how many bytes were written
57    ///
58    /// Unless this method errors, it waits until some amount of `buf` can be written into this
59    /// stream, and then writes as much as it can without waiting again. Due to congestion and flow
60    /// control, this may be shorter than `buf.len()`. On success this yields the length of the
61    /// prefix that was written.
62    ///
63    /// # Cancel safety
64    ///
65    /// This method is cancellation safe. If this does not resolve, no bytes were written.
66    pub async fn write(&mut self, buf: &[u8]) -> Result<usize, WriteError> {
67        poll_fn(|cx| self.execute_poll(cx, |s| s.write(buf))).await
68    }
69
70    /// Write a buffer into this stream in its entirety
71    ///
72    /// This method repeatedly calls [`write`](Self::write) until all bytes are written, or an
73    /// error occurs.
74    ///
75    /// # Cancel safety
76    ///
77    /// This method is *not* cancellation safe. Even if this does not resolve, some prefix of `buf`
78    /// may have been written when previously polled.
79    pub async fn write_all(&mut self, mut buf: &[u8]) -> Result<(), WriteError> {
80        while !buf.is_empty() {
81            let written = self.write(buf).await?;
82            buf = &buf[written..];
83        }
84        Ok(())
85    }

87    /// Writes [`Bytes`] from a slice of buffers into this stream.
88    ///
89    /// Returns how many bytes were written.
90    ///
91    /// Bytes to try to write are provided to this method as an array of cheaply cloneable chunks.
92    /// Unless this method errors, it waits until some amount of those bytes can be written into
93    /// this stream, and then writes as much as it can without waiting again. Due to congestion and
94    /// flow control, this may be less than the total number of bytes.
95    ///
96    /// On success, this method both mutates `bufs` and returns the number of bytes written:
97    ///
98    /// - `bufs` is advanced past chunks that were fully written.
99    /// - If a [`Bytes`] chunk was partially written, the chunk at the new front of `bufs` is [split
100    ///   to](Bytes::split_to) contain only the suffix of bytes that were not written.
101    ///
102    /// # Cancel safety
103    ///
104    /// This method is cancellation safe. If this does not resolve, no bytes were written.
105    pub async fn write_many_chunks(
106        &mut self,
107        bufs: &mut &mut [Bytes],
108    ) -> Result<usize, WriteError> {
109        poll_fn(|cx| self.execute_poll(cx, |s| s.write_chunks(bufs))).await
110    }
111
112    /// Writes a single [`Bytes`] into this stream in its entirety.
113    ///
114    /// Bytes to write are provided to this method as a single cheaply cloneable chunk. This
115    /// method repeatedly calls [`write_many_chunks`](Self::write_many_chunks) until all bytes
116    /// are written, or an error occurs.
117    ///
118    /// # Cancel safety
119    ///
120    /// This method is *not* cancellation safe. Even if this does not resolve, some bytes may have
121    /// been written when previously polled.
122    pub async fn write_chunk(&mut self, buf: Bytes) -> Result<(), WriteError> {
123        self.write_all_chunks(&mut [buf]).await
124    }
125
126    /// Writes a slice of [`Bytes`] into this stream in its entirety.
127    ///
128    /// Bytes to write are provided to this method as an array of cheaply cloneable chunks. This
129    /// method repeatedly calls [`write_many_chunks`](Self::write_many_chunks) until all bytes are
130    /// written, or an error occurs.
131    ///
132    /// # Cancel safety
133    ///
134    /// This method is *not* cancellation safe. Even if this does not resolve, some bytes may have
135    /// been written when previously polled.
136    pub async fn write_all_chunks(&mut self, bufs: &mut [Bytes]) -> Result<(), WriteError> {
137        let mut bufs = &mut bufs[..];
138        while !bufs.is_empty() {
139            self.write_many_chunks(&mut bufs).await?;
140        }
141        Ok(())
142    }

144    fn execute_poll<F, R>(
145        &mut self,
146        cx: &mut Context<'_>,
147        write_fn: F,
148    ) -> Poll<Result<R, WriteError>>
149    where
150        F: FnOnce(&mut proto::SendStream<'_>) -> Result<R, proto::WriteError>,
151    {
152        use proto::WriteError::*;
153        let mut conn = self.conn.lock_and_wake("SendStream::poll_write");
154        if self.is_0rtt && conn.check_0rtt().is_err() {
155            conn.skip_waking();
156            return Poll::Ready(Err(WriteError::ZeroRttRejected));
157        }
158        if let Some(conn_err) = conn.error.clone() {
159            conn.skip_waking();
160            return Poll::Ready(Err(WriteError::ConnectionLost(conn_err)));
161        }
162
163        let result = match write_fn(&mut conn.inner.send_stream(self.stream)) {
164            Ok(result) => result,
165            Err(Blocked) => {
166                conn.blocked_writers.insert(self.stream, cx.waker().clone());
167                conn.skip_waking();
168                return Poll::Pending;
169            }
170            Err(Stopped(error_code)) => {
171                conn.skip_waking();
172                return Poll::Ready(Err(WriteError::Stopped(error_code)));
173            }
174            Err(ClosedStream) => {
175                conn.skip_waking();
176                return Poll::Ready(Err(WriteError::ClosedStream));
177            }
178        };
179
180        Poll::Ready(Ok(result))
181    }

183    /// Notify the peer that no more data will ever be written to this stream
184    ///
185    /// It is an error to write to a [`SendStream`] after `finish()`ing it. [`reset()`](Self::reset)
186    /// may still be called after `finish` to abandon transmission of any stream data that might
187    /// still be buffered.
188    ///
189    /// To wait for the peer to receive all buffered stream data, see [`stopped()`](Self::stopped).
190    ///
191    /// May fail if [`finish()`](Self::finish) or [`reset()`](Self::reset) was previously
192    /// called. This error is harmless and serves only to indicate that the caller may have
193    /// incorrect assumptions about the stream's state.
194    pub fn finish(&mut self) -> Result<(), ClosedStream> {
195        let mut conn = self.conn.lock_and_wake("finish");
196        if let Err(e) = conn.inner.send_stream(self.stream).finish() {
197            conn.skip_waking();
198            match e {
199                FinishError::ClosedStream => Err(ClosedStream::default()),
200                // Harmless. If the application needs to know about stopped streams at this point,
201                // it should call `stopped`.
202                FinishError::Stopped(_) => Ok(()),
203            }
204        } else {
205            Ok(())
206        }
207    }

209    /// Close the send stream immediately.
210    ///
211    /// No new data can be written after calling this method. Locally buffered data is dropped, and
212    /// previously transmitted data will no longer be retransmitted if lost. If an attempt has
213    /// already been made to finish the stream, the peer may still receive all written data.
214    ///
215    /// May fail if [`finish()`](Self::finish) or [`reset()`](Self::reset) was previously
216    /// called. This error is harmless and serves only to indicate that the caller may have
217    /// incorrect assumptions about the stream's state.
218    pub fn reset(&mut self, error_code: VarInt) -> Result<(), ClosedStream> {
219        let mut conn = self.conn.lock_and_wake("SendStream::reset");
220        if self.is_0rtt && conn.check_0rtt().is_err() {
221            conn.skip_waking();
222            return Ok(());
223        }
224        conn.inner.send_stream(self.stream).reset(error_code)?;
225        Ok(())
226    }

228    /// Set the priority of the send stream
229    ///
230    /// Every send stream has an initial priority of 0. Locally buffered data from streams with
231    /// higher priority will be transmitted before data from streams with lower priority. Changing
232    /// the priority of a stream with pending data may only take effect after that data has been
233    /// transmitted. Using many different priority levels per connection may have a negative
234    /// impact on performance.
235    pub fn set_priority(&self, priority: i32) -> Result<(), ClosedStream> {
236        let mut conn = self.conn.lock_without_waking("SendStream::set_priority");
237        conn.inner.send_stream(self.stream).set_priority(priority)?;
238        Ok(())
239    }

241    /// Get the priority of the send stream
242    pub fn priority(&self) -> Result<i32, ClosedStream> {
243        let mut conn = self.conn.lock_without_waking("SendStream::priority");
244        conn.inner.send_stream(self.stream).priority()
245    }

247    /// Sets a new callback to set the [`StreamAffinity`].
248    ///
249    /// The stream affinity allows to restrict over which network paths the *application
250    /// data* for a stream is sent, i.e. it only affects STREAM frames. The callback is
251    /// invoked immediately upon setting it, as well as any time that the available network
252    /// paths or their properties change.
253    ///
254    /// Conditions where this callback will be invoked:
255    /// - A path is *opened*.
256    /// - A path is *established*.
257    /// - Packet scheduling stops using a path.
258    /// - Packet scheduling starts using a path.
259    /// - A path's status is changed (PATH_STATUS_AVAILABLE, PATH_STATUS_BACKUP).
260    /// - A path is *closed*.
261    ///
262    /// A path is *opened* whenever either the local side opens the path, or a packet was
263    /// received from the peer on the path. In the former case there is no implication yet
264    /// that the path will be able to carry any datagrams between the two peers. A path is
265    /// *established* when their is indication that it is able to carry datagrams between
266    /// the peers.
267    ///
268    /// Packet scheduling affects all QUIC frames and not only application data. It
269    /// functions entirely independent of stream affinity. It may decide to stop scheduling
270    /// packets on a path depending on many circumstances, e.g. because of increased packet
271    /// loss on a path, while the path might still be able to carry packets. When packet
272    /// scheduling stops using a path **no** packets are sent on that path, regardless of
273    /// what stream affinity indicates.
274    ///
275    /// Setting stream affinity to be sticky to a path on which no packets are sent means no
276    /// application data will be sent for the stream. This will quickly fill up the send
277    /// window and result in further writes being blocking.
278    ///
279    /// To react to gradual changes, e.g. a certain RTT threshold, paths must be monitored
280    /// outside of the callback and a new callback can be installed any time to change the
281    /// stream affinity.
282    pub fn set_affinity(
283        &mut self,
284        affinity_cb: Arc<dyn StreamAffinityCallback>,
285    ) -> Result<(), ClosedStream> {
286        self.affinity_cb = Some(affinity_cb);
287        let stream_priority = self.priority()?;
288        if let Some(ref cb) = self.affinity_cb {
289            let ctx = StreamAffinityContext {
290                conn: self.conn.clone(),
291                stream: self.stream,
292                stream_priority,
293            };
294            let affinity = cb.affinity(ctx);
295
296            let mut conn = self.conn.lock_without_waking("SendStream::priority");
297            conn.inner
298                .send_stream(self.stream)
299                .set_affinity(affinity.inner)?;
300        }
301        Ok(())
302    }

304    /// Completes when the peer stops the stream or reads the stream to completion
305    ///
306    /// Yields `Some` with the stop error code if the peer stops the stream. Yields `None` if the
307    /// local side [`finish()`](Self::finish)es the stream and then the peer acknowledges receipt
308    /// of all stream data (although not necessarily the processing of it), after which the peer
309    /// closing the stream is no longer meaningful.
310    ///
311    /// For a variety of reasons, the peer may not send acknowledgements immediately upon receiving
312    /// data. As such, relying on `stopped` to know when the peer has read a stream to completion
313    /// may introduce more latency than using an application-level response of some sort.
314    ///
315    /// Clients may wish to await this after finishing a unidirectional 0-RTT stream to reliably
316    /// determine whether the stream was rejected.
317    pub fn stopped(&self) -> Stopped {
318        let notified = {
319            // Create an `OwnedNotified` to move into the future. By creating it before the first
320            // poll, we make sure that we don't miss any notifications.
321            let mut conn = self.conn.lock_without_waking("SendStream::stopped");
322            conn.stopped
323                .entry(self.stream)
324                .or_default()
325                .clone()
326                .notified_owned()
327        };
328        Stopped {
329            conn: self.conn.clone(),
330            stream: self.stream,
331            is_0rtt: self.is_0rtt,
332            notified,
333        }
334    }

357/// Check if a send stream is stopped.
358///
359/// Returns `Some` if the stream is stopped or the connection is closed.
360/// Returns `None` if the stream is not stopped.
361fn send_stream_stopped(
362    conn: &mut State,
363    stream: StreamId,
364    is_0rtt: bool,
365) -> Option<Result<Option<VarInt>, StoppedError>> {
366    if is_0rtt && conn.check_0rtt().is_err() {
367        return Some(Err(StoppedError::ZeroRttRejected));
368    }
369    match conn.inner.send_stream(stream).stopped() {
370        Err(ClosedStream { .. }) => Some(Ok(None)),
371        Ok(Some(error_code)) => Some(Ok(Some(error_code))),
372        Ok(None) => conn.error.clone().map(|error| Err(error.into())),
373    }
374}

413impl Drop for SendStream {
414    fn drop(&mut self) {
415        let mut conn = self.conn.lock_and_wake("SendStream::drop");
416
417        // clean up any previously registered wakers
418        conn.blocked_writers.remove(&self.stream);
419
420        if conn.error.is_some() || (self.is_0rtt && conn.check_0rtt().is_err()) {
421            conn.skip_waking();
422            return;
423        }
424        match conn.inner.send_stream(self.stream).finish() {
425            Ok(()) => {}
426            Err(FinishError::Stopped(reason)) => {
427                if conn.inner.send_stream(self.stream).reset(reason).is_err() {
428                    conn.skip_waking()
429                }
430            }
431            // Already finished or reset, which is fine.
432            Err(FinishError::ClosedStream) => {
433                conn.skip_waking();
434            }
435        }
436    }
437}

458/// Errors that arise from writing to a stream
459#[derive(Debug, Error, Clone, PartialEq, Eq)]
460pub enum WriteError {
461    /// The peer is no longer accepting data on this stream
462    ///
463    /// Carries an application-defined error code.
464    #[error("sending stopped by peer: error {0}")]
465    Stopped(VarInt),
466    /// The connection was lost
467    #[error("connection lost")]
468    ConnectionLost(#[from] ConnectionError),
469    /// The stream has already been finished or reset
470    #[error("closed stream")]
471    ClosedStream,
472    /// This was a 0-RTT stream and the server rejected it
473    ///
474    /// Can only occur on clients for 0-RTT streams, which can be opened using
475    /// [`Connecting::into_0rtt()`].
476    ///
477    /// [`Connecting::into_0rtt()`]: crate::Connecting::into_0rtt()
478    #[error("0-RTT rejected")]
479    ZeroRttRejected,
480}
481
482impl From<ClosedStream> for WriteError {
483    #[inline]
484    fn from(_: ClosedStream) -> Self {
485        Self::ClosedStream
486    }
487}
488
489impl From<StoppedError> for WriteError {
490    fn from(x: StoppedError) -> Self {
491        match x {
492            StoppedError::ConnectionLost(e) => Self::ConnectionLost(e),
493            StoppedError::ZeroRttRejected => Self::ZeroRttRejected,
494        }
495    }
496}

498impl From<WriteError> for io::Error {
499    fn from(x: WriteError) -> Self {
500        use WriteError::*;
501        let kind = match x {
502            Stopped(_) | ZeroRttRejected => io::ErrorKind::ConnectionReset,
503            ConnectionLost(_) | ClosedStream => io::ErrorKind::NotConnected,
504        };
505        Self::new(kind, x)
506    }
507}

548impl Future for Stopped {
549    type Output = Result<Option<VarInt>, StoppedError>;
550
551    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
552        let mut this = self.project();
553        loop {
554            let mut conn = this.conn.lock_without_waking("SendStream::stopped");
555            // Check if the stream is stopped before polling the notify. This makes sure that
556            // no wakeups are missed.
557            if let Some(output) = send_stream_stopped(&mut conn, *this.stream, *this.is_0rtt) {
558                return Poll::Ready(output);
559            }
560            std::task::ready!(this.notified.as_mut().poll(cx));
561        }
562    }

565/// Trait to implement a callback to [set stream affinity].
566///
567/// [set stream affinity]: SendStream::set_affinity
568pub trait StreamAffinityCallback: Send + Sync {
569    /// Sets the affinity for a single stream.
570    ///
571    /// This function must not block.
572    ///
573    /// The provided [`StreamAffinityContext`] gives some information about the connection
574    /// and stream the callback is being invoked for as well as the current paths available
575    /// right now.
576    ///
577    /// The desired affinity for the stream should be returned by an instance of the
578    /// [`StreamAffinity`] struct. This struct can be created directly using one of the
579    /// public constructors.
580    fn affinity(&self, ctx: StreamAffinityContext) -> StreamAffinity;
581}

583/// Context for the [`StreamAffinityCallback`] being called.
584///
585/// This provides information for which connection and stream the callback is being
586/// called. It also allows access to the current paths available to the connection.
587#[derive(Debug)]
588pub struct StreamAffinityContext {
589    conn: ConnectionRef,
590    stream: StreamId,
591    /// The stream priority.
592    ///
593    /// Captured to avoid unwrap on [`ClosedStream`].
594    stream_priority: i32,
595}
596
597impl StreamAffinityContext {
598    /// The [stable ID](stable_id) of the connection this callback is being invoked for.
599    ///
600    /// [stable_id]: crate::Connection::stable_id
601    pub fn connection_stable_id(&self) -> usize {
602        self.conn.stable_id()
603    }
604
605    /// whether this is the server or client side of the connection.
606    pub fn connection_side(&self) -> Side {
607        self.conn
608            .lock_without_waking("StreamAffinityContext::connection_side")
609            .inner
610            .side()
611    }
612
613    /// Returns the identity of the stream.
614    pub fn stream_id(&self) -> StreamId {
615        self.stream
616    }
617
618    /// Returns the priority of the stream.
619    pub fn stream_priority(&self) -> i32 {
620        self.stream_priority
621    }
622
623    /// Iterator over all the currently known paths.
624    ///
625    /// This is a snapshot in time and should not be stored. The yielded [`Path`] objects
626    /// allow manipulation, it is encouraged to refrain from this and only inspect the
627    /// paths. Manipulation form here will make it harder to understand what is happening to
628    /// the connection.
629    pub fn paths(&self) -> impl Iterator<Item = Path> + '_ {
630        let paths = self
631            .conn
632            .lock_without_waking("Connection::iter_paths")
633            .path_refs
634            .iter()
635            .map(|(path_id, ref_owner)| {
636                Path::from_parts(self.conn.clone(), ref_owner.acquire(*path_id))
637            })
638            .collect::<Vec<_>>();
639        paths.into_iter()
640    }
641}

643/// Structure to allow choosing an affinity for a stream.
644///
645/// An instance of this needs to be returned in [`StreamAffinityCallback::affinity`].
646///
647/// The default affinity is [`StreamAffinity::any`], which is the implied affinity if no
648/// explicit stream affinity is set.
649#[derive(Debug, Default)]
650pub struct StreamAffinity {
651    inner: proto::StreamAffinity,
652}
653
654impl StreamAffinity {
655    /// Creates a new struct to return from [`StreamAffinityCallback::affinity`].
656    ///
657    /// Will use the default affinity, [`Self::any`].
658    pub fn new() -> Self {
659        Self::default()
660    }
661
662    /// Sets not specific affinity for this stream, the default behaviour.
663    ///
664    /// STREAM frames for this stream may be sent on any path that the packet scheduling
665    /// decides could carry the data to reach the peer.
666    pub fn any(&mut self) {
667        self.inner = proto::StreamAffinity::Any;
668    }
669
670    /// Marks this stream to be only transmitted on a particular network path.
671    pub fn sticky(&mut self, path: proto::PathId) {
672        self.inner = proto::StreamAffinity::Sticky(path);
673    }
674
675    /// Stops transmitting data for this stream.
676    ///
677    /// If no network path is suitable to carry this stream's data it is possible to not
678    /// send it at all. This will result in write calls on the [`SendStream`] blocking once
679    /// the send window is filled up.
680    pub fn block(&mut self) {
681        self.inner = proto::StreamAffinity::Block;
682    }
683}
